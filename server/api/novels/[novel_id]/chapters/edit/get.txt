import { defineEventHandler } from 'h3';
import { verifyUser, getUserRole } from '~/server/utils';
import prisma from "~/prisma/middleware/index.js";

export default defineEventHandler(async (event) => {
  const { novel_id } = event.context.params;
  const { query, page = 1, limit = 20 } = getQuery(event); // Default page is 1, limit is 20
  const isVerified = verifyUser(event);

  if (!isVerified) {
    return createError({
      statusCode: 401,
      message: "You are not logged in to use this feature",
    });
  }

  const userId = event.context.session.user.id;
  
  try {
    // Verify if the user is the writer or uploader of the novel
    const userRole = await getUserRole(userId, novel_id);
    if (userRole !== 'writer' && userRole !== 'uploader') {
      return {
        statusCode: 403,
        message: 'Forbidden: You are not authorized to view this chapter.',
      };
    }

    // Determine the target collection based on the query
    let targetCollection;
    switch (query) {
      case 'draft':
        targetCollection = prisma.drafts;
        break;
      case 'unpub':
        targetCollection = prisma.unPublishedChapter;
        break;
      case 'pub':
        targetCollection = prisma.chapter;
        break;
      default:
        return {
          statusCode: 400,
          message: 'Invalid query parameter.',
        };
    }

    const pageNumber = parseInt(page, 10);
    const pageSize = parseInt(limit, 10);
    const skip = (pageNumber - 1) * pageSize;

    // Fetch chapters with pagination
    const chapters = await targetCollection.findMany({
      where: {
        novelId: parseInt(novel_id),
      },
      skip: skip,
      take: pageSize,
      orderBy: {
        id: 'asc', // Modify this based on your sorting requirements
      },
    });

    // Count total chapters for pagination metadata
    const totalChapters = await targetCollection.count({
      where: {
        novelId: parseInt(novel_id),
      },
    });

    if (!chapters.length) {
      return {
        statusCode: 404,
        message: 'No chapters found.',
      };
    }

    return {
      statusCode: 200,
      body: {
        chapters,
        pagination: {
          totalItems: totalChapters,
          totalPages: Math.ceil(totalChapters / pageSize),
          currentPage: pageNumber,
          pageSize,
        },
      },
    };
  } catch (error) {
    console.error('Error fetching chapters:', error);
    return {
      statusCode: 500,
      body: { message: 'Internal server error.' },
    };
  } 
});
