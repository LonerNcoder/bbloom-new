<template>
  <div class="review-section">
    <!-- Add Review Button -->
    <button 
      v-if="!hasReviewed" 
      @click="showAddForm = true" 
      class="add-review-btn"
    >
      Add Review
    </button>

    <!-- Add Review Form -->
    <div v-if="showAddForm" class="add-review">
      <div class="rating">
        <span 
          v-for="star in 5" 
          :key="star" 
          @click="setRating(star)"
          :class="{ 'selected': newReview.rating >= star }"
        >
          ★
        </span>
      </div>
      <textarea 
        v-model="newReview.content" 
        placeholder="Write your review..."
      ></textarea>
      <button @click="submitReview">Submit Review</button>
      <button @click="showAddForm = false">Cancel</button>
    </div>

    <!-- Empty State -->
    <div v-if="reviews.length === 0" class="empty-state">
      No reviews yet. Be the first to review!
    </div>

    <!-- Reviews List -->
    <div v-for="review in reviews" :key="review.id" class="review">
      <div class="review-header">
        <div class="review-meta">
          <span class="username">{{ review.username }}</span>
          <span class="rating">
            {{ '★'.repeat(review.rating) }}{{ '☆'.repeat(5 - review.rating) }}
          </span>
          <small>{{ new Date(review.createdAt).toLocaleString() }}</small>
        </div>
        <div class="review-voting">
          <button @click="voteReview(review.id, 'upvote')">
            ↑ {{ review.upvotes || 0 }}
          </button>
          <button @click="voteReview(review.id, 'downvote')">
            ↓ {{ review.downvotes || 0 }}
          </button>
        </div>
      </div>

      <div class="content">{{ review.content }}</div>

      <!-- Comments Section -->
      <div class="comments">
        <!-- Comment Input -->
        <div class="comment-form">
          <textarea 
            v-model="newComments[review.id]"
            placeholder="Write a comment..."
          ></textarea>
          <button @click="submitComment(review.id)">
            Post Comment
          </button>
        </div>

        <!-- Comments Recursive Component -->
        <CommentThread 
          :comments="review.comments" 
          :review-id="review.id"
          @reply="handleReply"
        />
      </div>
    </div>
  </div>
  </template>
  
  <script setup>
  import { ref, computed, watch } from 'vue';

  
  const { $store } = useNuxtApp();
  const props = defineProps({ instanceId: { type: String, required: true } });
  
  // Reactive state
  const reviews = ref([]);
  const showAddForm = ref(false);
  const newReview = ref({ rating: 0, content: '' });
  const newComments = ref({});
  const currentUserId = await $store.getUserId();;
  const hasReviewed = computed(() => 
    reviews.value.some(r => r.userId === currentUserId)
  );
  
  // Fetch reviews
  watch(() => props.instanceId, async (instanceId) => {
    if (!instanceId) return;
    
    try {
      const response = await $fetch(`${API}reviews/${instanceId}`, {
        headers: await $store.getNormalHeaders()
      });
      reviews.value = response.data.map(r => ({
        ...r,
        comments: r.comments.map(c => ({
          ...c,
          replies: c.replies || []
        }))
      }));
    } catch (error) {
      console.error('Error fetching reviews:', error);
    }
  }, { immediate: true });
  
  // Review actions
  const setRating = (stars) => newReview.value.rating = stars;
  
  const submitReview = async () => {
    try {
      const response = await $fetch(`${API}reviews/${props.instanceId}`, {
        method: "POST",
        headers: await $store.getNormalHeaders(),
        body: newReview.value
      });
      
      reviews.value = [response, ...reviews.value];
      showAddForm.value = false;
      newReview.value = { rating: 0, content: '' };
    } catch (error) {
      console.error('Error submitting review:', error);
    }
  };
  
  const voteReview = async (reviewId, type) => {
    try {
      const response = await $fetch(`${API}reviews/${reviewId}/vote`, {
        method: "POST",
        headers: await $store.getNormalHeaders(),
        body: { type }
      });
  
      reviews.value = reviews.value.map(review => 
        review.id === reviewId ? { 
          ...review, 
          upvotes: response.upvotes, 
          downvotes: response.downvotes 
        } : review
      );
    } catch (error) {
      console.error('Error voting:', error);
    }
  };
  
  // Comment actions
  const submitComment = async (reviewId) => {
    try {
      const content = newComments.value[reviewId]?.trim();
      if (!content) return;
  
      const response = await $fetch(`${API}reviews/comment/${reviewId}/reply`, {
        method: "POST",
        headers: await $store.getNormalHeaders(),
        body: { content }
      });
      console.log(response)
  
      reviews.value = reviews.value.map(review => 
        review.id === reviewId ? { 
          ...review, 
          comments: [response, ...review.comment] 
        } : review
      );
      
      newComments.value[reviewId] = '';
    } catch (error) {
      console.error('Error submitting comment:', error);
    }
  };
  
  const handleReply = async ({ parentId, content }) => {
    try {
      const response = await $fetch(`${API}reviews/comment/${parentId}/reply`, {
        method: "POST",
        headers: await $store.getNormalHeaders(),
        body: { content }
      });
  
      const updateNestedComments = (comments) => comments.map(comment => {
        if (comment.id === parentId) {
          return { ...comment, replies: [...comment.replies, response] };
        }
        return comment.replies?.length ? {
          ...comment,
          replies: updateNestedComments(comment.replies)
        } : comment;
      });
  
      reviews.value = reviews.value.map(review => ({
        ...review,
        comments: updateNestedComments(review.comments)
      }));
    } catch (error) {
      console.error('Error submitting reply:', error);
    }
  };
  </script>
