import { useState, useEffect } from 'react';

const BookLoadingAnimation = () => {
  const [currentFact, setCurrentFact] = useState('');
  const [showFact, setShowFact] = useState(false);
  const [isOpen, setIsOpen] = useState(false);

  const writerFacts = [
    "Shakespeare used 28,829 unique words in his works",
    "Agatha Christie wrote 66 detective novels",
    "J.K. Rowling was rejected by 12 publishers",
    "Ernest Hemingway wrote standing up",
    "Victor Hugo wrote Les Misérables in the nude",
    "Maya Angelou worked as a cable car conductor",
    "Charles Dickens had a secret door in his study"
  ];

  useEffect(() => {
    setTimeout(() => setIsOpen(true), 300);
    
    const randomFact = writerFacts[Math.floor(Math.random() * writerFacts.length)];
    const timer = setTimeout(() => {
      setCurrentFact(randomFact);
      setShowFact(true);
    }, 800);

    return () => {
      clearTimeout(timer);
      setShowFact(false);
      setIsOpen(false);
    };
  }, []);

  // Generate multiple page elements with different animations
  const renderPages = () => {
    return Array.from({ length: 12 }).map((_, index) => (
      <div
        key={index}
        className={`absolute left-1/2 top-1/2 w-32 md:w-40 h-4/5 bg-white shadow-sm origin-left
          ${isOpen ? 'page-open' : 'page-closed'}
          page-${index + 1}`}
        style={{
          zIndex: 20 - index,
          transformOrigin: 'left center',
          animationDelay: `${index * 0.15}s`
        }}
      >
        {/* Page content with lines */}
        <div className="h-full bg-white p-4">
          <div className="h-full bg-gradient-to-l from-gray-100 to-transparent opacity-80">
            {/* Decorative text lines */}
            {Array.from({ length: 6 }).map((_, lineIndex) => (
              <div
                key={lineIndex}
                className="h-2 bg-gray-100 rounded my-2"
                style={{
                  width: `${Math.random() * 30 + 60}%`,
                  opacity: 0.5
                }}
              />
            ))}
          </div>
        </div>
      </div>
    ));
  };

  return (
    <div className="flex flex-col items-center justify-center min-h-screen bg-gradient-to-b from-blue-50 to-white">
      <div className="relative w-48 h-48 md:w-64 md:h-64">
        {/* Book cover - back */}
        <div className={`absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2 w-64 md:w-80 h-4/5 
          bg-blue-800 rounded-lg shadow-xl transform 
          ${isOpen ? 'book-cover-open' : 'book-cover-closed'}`} />
        
        {/* Book spine */}
        <div className="absolute left-1/2 top-1/2 -translate-x-1/2 -translate-y-1/2 w-8 h-4/5 
          bg-blue-700 rounded-sm shadow-lg">
          <div className="absolute inset-x-0 top-4 bottom-4 flex flex-col justify-between">
            <div className="w-full h-px bg-blue-500" />
            <div className="w-full h-px bg-blue-500" />
          </div>
        </div>
        
        {/* Flipping pages */}
        {renderPages()}
        
        {/* Book cover - front */}
        <div className={`absolute left-1/2 top-1/2 -translate-x-full -translate-y-1/2 w-32 md:w-40 h-4/5 
          bg-blue-700 rounded-l-lg shadow-xl transform origin-right
          ${isOpen ? 'cover-open' : 'cover-closed'}`}>
          <div className="absolute inset-0 bg-gradient-to-r from-blue-900 to-blue-700">
            {/* Decorative cover design */}
            <div className="absolute inset-4 border-2 border-blue-500 rounded opacity-30" />
          </div>
        </div>
      </div>
      
      {/* Loading text */}
      <div className="mt-8 text-center">
        <h2 className="text-2xl font-bold text-blue-700 mb-2 animate-pulse">
          Opening Pages...
        </h2>
        
        <div className={`max-w-md px-6 py-3 text-sm text-gray-600 transition-opacity duration-1000 
          ${showFact ? 'opacity-100' : 'opacity-0'}`}>
          <p className="italic">Did you know?</p>
          <p className="mt-1">{currentFact}</p>
        </div>
      </div>

      <style jsx>{`
        .book-cover-closed {
          transform: translate(-50%, -50%) rotateX(0deg);
        }
        
        .book-cover-open {
          transform: translate(-50%, -50%) rotateX(-10deg);
        }
        
        .cover-closed {
          transform: translateX(-100%) translateY(-50%) rotateY(0deg);
        }
        
        .cover-open {
          transform: translateX(-100%) translateY(-50%) rotateY(-120deg);
          transition: transform 1.2s cubic-bezier(0.4, 0, 0.2, 1);
        }
        
        .page-closed {
          transform: translateY(-50%) rotateY(0deg);
        }
        
        .page-open {
          animation: pageFlip 3s ease-in-out infinite;
        }
        
        ${Array.from({ length: 12 }).map((_, i) => `
          .page-${i + 1} {
            animation-delay: ${i * 0.15}s;
          }
        `).join('\n')}
        
        @keyframes pageFlip {
          0% {
            transform: translateY(-50%) rotateY(0deg);
          }
          20% {
            transform: translateY(-50%) rotateY(-120deg);
          }
          35% {
            transform: translateY(-50%) rotateY(-90deg);
          }
          50% {
            transform: translateY(-50%) rotateY(-110deg);
          }
          65% {
            transform: translateY(-50%) rotateY(-95deg);
          }
          80% {
            transform: translateY(-50%) rotateY(-105deg);
          }
          100% {
            transform: translateY(-50%) rotateY(-180deg);
          }
        }
      `}</style>
    </div>
  );
};

export default BookLoadingAnimation;
